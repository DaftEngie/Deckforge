# Recompute every number in CLEAN_SLATE_NATIVE.md section 7 and related arithmetic.
from math import sqrt, tan, radians
from statistics import NormalDist

def fps(ms): return 1000.0 / ms

def stack(base, v1_frac, sweep_frac=0.0, extra_ms=0.0):
    v1 = base * v1_frac
    rem = base - v1
    sw = rem * sweep_frac
    ft = rem - sw - extra_ms
    return v1, sw, ft, fps(ft)

print("== Wukong High-like, doc's numbers (54 ms base) ==")
for name, args in [("pess 3%", (0.03,)), ("cent 12% + sweep 1.5%", (0.12, 0.015)),
                   ("opt 25% + sweep 4% + HalfRate 1.4", (0.25, 0.04, 1.4))]:
    v1, sw, ft, f = stack(54.0, *args)
    print(f"  {name:36s} V1={v1:5.2f} ms sweep={sw:4.2f} ms -> {ft:5.2f} ms = {f:5.2f} fps; gain x{54/ft:4.2f}")

print("== Wukong using E_F note's own range (3 / 8-12 / 20 %), no Half Rate on top ==")
for name, args in [("pess 3%", (0.03,)), ("cent 10% + sweep 1.5%", (0.10, 0.015)),
                   ("opt 20% + sweep 4%", (0.20, 0.04))]:
    v1, sw, ft, f = stack(54.0, *args)
    print(f"  {name:36s} -> {ft:5.2f} ms = {f:5.2f} fps; gain x{54/ft:4.2f}")

print("== Medium-plus Wukong: 31.25 ms minus 1/3/6 ms ==")
for s in (1, 3, 6):
    print(f"  -{s} ms -> {31.25-s:5.2f} ms = {fps(31.25-s):5.2f} fps")
print("  clean-lock average needed for 1%-low ratio 0.75/0.80/0.85:",
      [round(30/r, 1) for r in (0.75, 0.80, 0.85)])
print("  saving needed to hit 37.5 fps from 31.25 ms:", round(31.25 - 1000/37.5, 2), "ms")
print("  W1 Q3 saving after R4 cut mip bias 0.6->0.2 ms: 3-4 ms becomes", 3-0.4, "-", 4-0.4, "ms")

print("== Stalker 2 (53.5 ms) same method ==")
for name, args in [("pess 3%", (0.03,)), ("cent 12% + sweep 1.5%", (0.12, 0.015)),
                   ("opt 25% + sweep 4% + HalfRate 1.4", (0.25, 0.04, 1.4))]:
    v1, sw, ft, f = stack(53.5, *args)
    print(f"  {name:36s} -> {ft:5.2f} ms = {f:5.2f} fps")

print("== Wilds (67 ms): menu ~10% + Half Rate 2.4 ms ==")
_, _, ft, f = stack(67.0, 0.10, 0.0, 2.4); print(f"  menu+HR only          -> {ft:5.2f} ms = {f:5.2f} fps")
_, _, ft, f = stack(67.0, 0.10, 0.015, 2.4); print(f"  menu+HR+sweep 1.5%    -> {ft:5.2f} ms = {f:5.2f} fps")

print("== Space Marine 2 hordes ~18 fps ==")

    print(f"  +{g*100:4.1f}% -> {18*(1+g):5.2f} fps")
